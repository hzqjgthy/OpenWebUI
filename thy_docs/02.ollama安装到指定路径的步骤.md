
# 0. 激活环境
conda activate OpenWebUI

# 1. 创建目录
mkdir -p /root/autodl-tmp/ollama

# 2. 设置环境变量
export OLLAMA_MODELS="/root/autodl-tmp/ollama"

# 3. 永久保存（添加到配置文件）
echo 'export OLLAMA_MODELS="/root/autodl-tmp/ollama"' >> ~/.bashrc
source ~/.bashrc

# 这里需要再次激活环境
conda activate OpenWebUI

# 安装gpu检测工具
apt-get update && apt-get install -y pciutils lshw

# 4. 安装或重启 Ollama
curl -fsSL https://ollama.com/install.sh | sh
# 或者如果已安装：
sudo systemctl restart ollama

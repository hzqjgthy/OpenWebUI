
安装 OpenWebUI
pip install open-webui

设置离线环境，避免Open-WebUI启动时自动进行模型下载
export HF_HUB_OFFLINE=1

启动Open-WebUI
open-webui serve

                启动 MCPO 代理（新开一个终端）
                        # 安装 uvx（如果还没有）
                        apt install -y pipx
                        pipx ensurepath
                        source ~/.bashrc

                        # 或者直接使用 pip 安装 uv
                        pip install uv

                        # 运行 MCPO 代理
                        uvx mcpo --port 9000 --api-key "your-secret-key-123" --server-type "streamable-http" -- http://localhost:3000/mcp

                        # 端口映射 9000

                        在openwebui前端配置url：http://localhost:9000
